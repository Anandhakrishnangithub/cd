#include <stdio.h>
#include <string.h>
int main() {
    char op[5], arg1[10], arg2[10], result[10];
    FILE *fp1, *fp2;
    fp1 = fopen("input.txt", "r");
    fp2 = fopen("output.txt", "w");
    if (fp1 == NULL) {
        printf("Error: input.txt not found.\n");
        return 1;
    }
    while (fscanf(fp1, "%s %s %s %s", op, arg1, arg2, result) == 4) {
        if (strcmp(op, "+") == 0) {
            fprintf(fp2, "MOV AX, %s\n", arg1);
            fprintf(fp2, "ADD AX, %s\n", arg2);
            fprintf(fp2, "MOV %s, AX\n\n", result);
        } 
        else if (strcmp(op, "-") == 0) {
            fprintf(fp2, "MOV AX, %s\n", arg1);
            fprintf(fp2, "SUB AX, %s\n", arg2);
            fprintf(fp2, "MOV %s, AX\n\n", result);
        } 
        else if (strcmp(op, "*") == 0) {
            fprintf(fp2, "MOV AX, %s\n", arg1);
            fprintf(fp2, "MOV BX, %s\n", arg2);
            fprintf(fp2, "MUL BX\n");
            fprintf(fp2, "MOV %s, AX\n\n", result);
        } 
        else if (strcmp(op, "/") == 0) {
            fprintf(fp2, "MOV AX, %s\n", arg1);
            fprintf(fp2, "MOV BX, %s\n", arg2);
            fprintf(fp2, "DIV BX\n");
            fprintf(fp2, "MOV %s, AX\n\n", result);
        } 
        else if (strcmp(op, "=") == 0) {
            fprintf(fp2, "MOV AX, %s\n", arg1);
            fprintf(fp2, "MOV %s, AX\n\n", result);
        }
    }

    fclose(fp1);
    fclose(fp2);
    printf("8086 Target Assembly Code generated in output.txt\n");
    return 0;
}
